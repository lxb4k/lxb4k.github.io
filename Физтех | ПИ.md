  (function initSchedule() {
    // Quartz/Obsidian page: remember the user's group and week locally.
    let currentGroup = localStorage.getItem('userGroup') || '262';
    let currentWeek = localStorage.getItem('userWeek') || getAutoWeek();
    let deferredPrompt = null;

    function getAutoWeek() {
      const now = new Date();
      const startOfYear = new Date(now.getFullYear(), 0, 1);
      const weekNumber = Math.ceil((((now - startOfYear) / 86400000) + startOfYear.getDay() + 1) / 7);
      return (weekNumber % 2 === 0) ? 'b' : 'a';
    }

    window.switchTab = function(tabName, btn) {
      document.querySelectorAll('.nav-tab-btn').forEach(b => b.classList.remove('active'));
      document.querySelectorAll('.tab-content').forEach(c => c.classList.remove('active'));

      if (btn) btn.classList.add('active');
      const targetTab = document.getElementById(`tab-${tabName}`);
      if (targetTab) targetTab.classList.add('active');
      localStorage.setItem('activeTab', tabName);
    };

    function updateDisplay() {
      document.querySelectorAll('.schedule-block').forEach(el => el.classList.remove('active'));

      document.querySelectorAll('.group-btn').forEach(btn => {
        const selected = btn.dataset.group === currentGroup;
        btn.classList.toggle('active', selected);
        btn.setAttribute('aria-pressed', String(selected));
      });

      document.querySelectorAll('.week-btn').forEach(btn => {
        const selected = btn.dataset.week === currentWeek;
        btn.classList.toggle('active', selected);
        btn.setAttribute('aria-pressed', String(selected));
      });

      const targetId = `sched-${currentGroup}-${currentWeek}`;
      const target = document.getElementById(targetId);
      if (target) {
        target.classList.add('active');
        highlightTodayAndLesson(target);
      }
    }

    window.setGroup = function(group) {
      currentGroup = group;
      localStorage.setItem('userGroup', group);
      updateDisplay();
    };

    window.setWeek = function(week) {
      currentWeek = week;
      localStorage.setItem('userWeek', week);
      updateDisplay();
    };

    function highlightTodayAndLesson(container) {
      const now = new Date();
      const dayOfWeek = now.getDay(); // 0 = Sun, 1 = Mon ... 6 = Sat
      const currentMins = now.getHours() * 60 + now.getMinutes();

      container.querySelectorAll('.day-column').forEach(c => c.classList.remove('today'));
      container.querySelectorAll('.lesson-card').forEach(c => {
        c.classList.remove('active-lesson', 'next-lesson');
      });

      const label = document.getElementById('current-day-label');
      if (label) {
        const names = ['Воскресенье', 'Понедельник', 'Вторник', 'Среда', 'Четверг', 'Пятница', 'Суббота'];
        const groupName = currentGroup === '262' ? 'ПИ-б-о-262' : 'ПИ-б-о-261';
        label.textContent = `📅 Сегодня: ${names[dayOfWeek]} · ${groupName}`;
      }

      if (dayOfWeek === 0) return;

      const todayCol = container.querySelector(`.day-column[data-day="${dayOfWeek}"]`);
      if (!todayCol) return;

      todayCol.classList.add('today');

      // Quartz SPA navigation can reuse the same DOM, so do not reload the page.
      const cards = [...todayCol.querySelectorAll('.lesson-card[data-start][data-end]')];
      let activeFound = false;
      let nextCard = null;

      cards.forEach(card => {
        const [sH, sM] = card.dataset.start.split(':').map(Number);
        const [eH, eM] = card.dataset.end.split(':').map(Number);
        const startMins = sH * 60 + sM;
        const endMins = eH * 60 + eM;

        if (currentMins >= startMins && currentMins <= endMins) {
          card.classList.add('active-lesson');
          activeFound = true;
        } else if (!activeFound && startMins > currentMins && !nextCard) {
          nextCard = card;
        }
      });

      if (!activeFound && nextCard) {
        nextCard.classList.add('next-lesson');
      }

      // Only auto-scroll on mobile; this avoids Quartz desktop page jumps.
      if (window.matchMedia('(max-width: 768px)').matches) {
        setTimeout(() => {
          todayCol.scrollIntoView({
            behavior: 'smooth',
            block: 'nearest',
            inline: 'center'
          });
        }, 50);
      }
    }

    window.addEventListener('beforeinstallprompt', (e) => {
      e.preventDefault();
      deferredPrompt = e;
    });

    window.installPWA = function() {
      if (deferredPrompt) {
        deferredPrompt.prompt();
        deferredPrompt.userChoice.finally(() => { deferredPrompt = null; });
      } else {
        alert('Установка на телефон:\n\n• iPhone: Safari → «Поделиться» → «На экран “Домой”».\n• Android: Chrome → ⋮ → «Добавить на главный экран» / «Установить приложение».');
      }
    };

    // Quartz 4: service worker is a static resource from quartz/static/sw.js.
    if ('serviceWorker' in navigator) {
      window.addEventListener('load', () => {
        navigator.serviceWorker.register('./sw.js').catch(() => {});
      });
    }

    // Keep the current lesson highlight fresh without reloading Quartz.
    setInterval(() => {
      const active = document.querySelector('.schedule-block.active');
      if (active) highlightTodayAndLesson(active);
    }, 30000);

    // Quartz 4 SPA navigation: rebind the current schedule after every nav.
    document.addEventListener('nav', () => {
      updateDisplay();
      const savedTab = localStorage.getItem('activeTab');
      if (savedTab) {
        const btn = document.querySelector(`.nav-tab-btn[onclick*="${savedTab}"]`);
        if (btn) switchTab(savedTab, btn);
      }
    });

    if (document.readyState === 'loading') {
      document.addEventListener('DOMContentLoaded', () => {
        updateDisplay();
        const savedTab = localStorage.getItem('activeTab');
        if (savedTab) {
          const btn = document.querySelector(`.nav-tab-btn[onclick*="${savedTab}"]`);
          if (btn) switchTab(savedTab, btn);
        }
      });
    } else {
      updateDisplay();
    }
  })();
</script>
